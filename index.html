// ===== LÓGICA DE PLANTÃO (COM DISTRIBUIÇÃO EM BLOCO) ===== function getWorkersForDate(year, month, day) { let activeWorkers = []; let isDuty = false; const currentDate = new Date(year, month, day); const dayOfWeek = currentDate.getDay(); const isWeekday = dayOfWeek >= 1 && dayOfWeek <= 5; // REGRA 1: ALTERNÂNCIA (segunda a sexta) if (isWeekday) { const diffDays = daysBetween(currentDate, ALTERNANCIA_REF); if (diffDays % 2 === 0) isDuty = true; } // REGRA 2: FIM DE SEMANA (sexta, sábado e domingo) com rodízio em bloco if (dayOfWeek === 5 || dayOfWeek === 6 || dayOfWeek === 0) { const anchor = new Date(2026, 6, 3); const diff = daysBetween(currentDate, anchor); const pos = mathematicalMod(diff, 14); let isWeekendPlantao = false; if (pos === 7 && dayOfWeek === 5) isWeekendPlantao = true; // sexta else if (pos === 8 && dayOfWeek === 6) isWeekendPlantao = true; // sábado else if (pos === 9 && dayOfWeek === 0) isWeekendPlantao = true; // domingo if (isWeekendPlantao) { isDuty = true; const rotation = staffConfig.rotation || []; const qtd = staffConfig.quantidades || { sexta:1, sabado:1, domingo:1 }; const qtdSexta = Math.max(0, parseInt(qtd.sexta) || 1); const qtdSabado = Math.max(0, parseInt(qtd.sabado) || 1); const qtdDomingo = Math.max(0, parseInt(qtd.domingo) || 1); const totalPorFDS = qtdSexta + qtdSabado + qtdDomingo; if (rotation.length > 0 && totalPorFDS > 0) { // Número do fim de semana desde a referência (cada 14 dias tem um bloco) const weekendIdx = Math.floor((diff - pos) / 14); // Offset base para este fim de semana let offsetBase = mathematicalMod(weekendIdx * totalPorFDS, rotation.length); // Função para pegar uma fatia com wrap-around const getSlice = (start, count) => { const slice = []; for (let i = 0; i < count; i++) { const idx = mathematicalMod(start + i, rotation.length); slice.push(rotation[idx]); } return slice; }; // Distribuição sequencial let offsetSexta = offsetBase; let offsetSabado = mathematicalMod(offsetBase + qtdSexta, rotation.length); let offsetDomingo = mathematicalMod(offsetBase + qtdSexta + qtdSabado, rotation.length); // Adiciona os nomes conforme o dia if (dayOfWeek === 5) { activeWorkers.push(...getSlice(offsetSexta, qtdSexta)); } else if (dayOfWeek === 6) { activeWorkers.push(...getSlice(offsetSabado, qtdSabado)); } else if (dayOfWeek === 0) { activeWorkers.push(...getSlice(offsetDomingo, qtdDomingo)); } } } } // EXCEÇÕES (coringas) – só se o dia for plantão if (isDuty) { const excecoes = staffConfig.excecoes || {}; const excecaoNomes = excecoes[dayOfWeek] || []; if (excecaoNomes.length > 0) { activeWorkers.push(...excecaoNomes); } } // Remove duplicatas (caso haja alguma exceção que coincida) activeWorkers = [...new Set(activeWorkers)]; return { isDuty, activeWorkers }; }
